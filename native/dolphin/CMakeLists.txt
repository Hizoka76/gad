cmake_minimum_required(VERSION 3.16)
project(GadDolphinWatchdog LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus Xml)
find_package(KF6XmlGui REQUIRED)
add_library(gad_dolphin_watchdog MODULE plugin.cpp watchdog.h)
target_link_libraries(gad_dolphin_watchdog PRIVATE Qt6::Widgets Qt6::DBus KF6::XmlGui)
target_compile_options(gad_dolphin_watchdog PRIVATE -Wall -Wextra)
set_target_properties(gad_dolphin_watchdog PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/..")
option(GAD_WATCHDOG_TESTS "Build isolated watchdog tests" OFF)
if(GAD_WATCHDOG_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_library(watchdog-real-probe MODULE ../../tests/dolphin_watchdog_probe.cpp)
    target_link_libraries(watchdog-real-probe PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test KF6::XmlGui)
    add_executable(watchdog-test ../../tests/dolphin_watchdog.cpp watchdog.h)
    target_link_libraries(watchdog-test PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test KF6::XmlGui)
endif()
